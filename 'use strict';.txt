function normalizarTexto(txt) {
    if (!txt) return "";
    return txt.toString().toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").trim();
}

const LOCALES = LOCALES_BASE.map(local => {
    const aliasSet = new Set(local.alias ? local.alias.map(normalizarTexto) : []);
    const nombreLimpio = normalizarTexto(local.nombre);
    aliasSet.add(nombreLimpio);

    return {
        ...local,
        nombre: nombreLimpio,
        alias: Array.from(aliasSet)
    };
});

// 🛡️ ESTRUCTURAS TTL
// ==========================================
class TTLSet {
    constructor(ttlMs) { this.ttl = ttlMs; this.store = new Map(); }
    add(value) { this.store.set(value, Date.now()); }
    has(value) {
        const time = this.store.get(value);
        if (!time) return false;
        if (Date.now() - time > this.ttl) { this.store.delete(value); return false; }
        return true;
    }
}

class TTLMap {
    constructor(ttlMs) { this.ttl = ttlMs; this.store = new Map(); }
    set(key, value) { this.store.set(key, { value, timestamp: Date.now() }); }
    get(key) {
        const entry = this.store.get(key);
        if (!entry) return undefined;
        if (Date.now() - entry.timestamp > this.ttl) { this.store.delete(key); return undefined; }
        return entry.value;
    }
    has(key) { return this.get(key) !== undefined; }
}

function calcularDistancia(lat1, lng1, lat2, lng2) {
    const R = 6371;
    const dLat = (lat2 - lat1) * Math.PI / 180;
    const dLng = (lng2 - lng1) * Math.PI / 180;
    const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
            Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * 
            Math.sin(dLng/2) * Math.sin(dLng/2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    return parseFloat((R * c).toFixed(2));
}

async function obtenerNombreGrupo(sock, jid) {
    const cacheado = estado.cacheNombresGrupos.get(jid);
    if (cacheado) return cacheado;
    try {
        const metadata = await sock.groupMetadata(jid);
        const nombre = metadata.subject || jid;
        estado.cacheNombresGrupos.set(jid, nombre);
        return nombre;
    } catch (e) {
        return jid;
    }
}

function extraerTextoYUbicacion(msg) {
    const mensajeReal = msg.message.ephemeralMessage?.message ||
                        msg.message.viewOnceMessage?.message || 
                        msg.message.viewOnceMessageV2?.message || 
                        msg.message;
                        
    let texto = (mensajeReal.conversation || 
                mensajeReal.extendedTextMessage?.text || 
                mensajeReal.imageMessage?.caption || 
                mensajeReal.documentMessage?.caption ||
                mensajeReal.locationMessage?.name ||           
                mensajeReal.locationMessage?.address || 
                mensajeReal.liveLocationMessage?.caption || "").trim();

    let coords = null;
    if (mensajeReal.locationMessage || mensajeReal.liveLocationMessage) {
        const loc = mensajeReal.locationMessage || mensajeReal.liveLocationMessage;
        coords = { lat: loc.degreesLatitude, lng: loc.degreesLongitude };
    }

    if (!coords && texto) {
        const regexCoords = /@(-?\d+\.\d+),(-?\d+\.\d+)/;
        const match = texto.match(regexCoords);
        if (match) {
            coords = { lat: parseFloat(match[1]), lng: parseFloat(match[2]) };
        }
    }

    return { texto, coords, mensajeReal };
}

let botIniciado = false; // ✅ NUEVO: Control para evitar duplicados

async function iniciarBot() {
    if (botIniciado) {
        console.log("⚠️ El bot ya está iniciado. Ignorando nueva instancia.");
        return;
    }
    botIniciado = true;

    console.log("🤖 Iniciando motor Baileys...");
    const { state, saveCreds } = await useMultiFileAuthState('sesion_baileys');
    
    const sock = makeWASocket({
        logger: pino({ level: 'silent' }),
        auth: state,
        browser: ['Radar Cusco', 'Chrome', '1.0.0']
    });

    sock.ev.on('creds.update', saveCreds);

    sock.ev.on('connection.update', (update) => {
        const { connection, lastDisconnect, qr } = update;
        
        if (qr) {
            console.clear();
            console.log('✨ Escanea este QR para conectar el radar:\n');
            qrcode.generate(qr, { small: true });
        }

        if (connection === 'close') {
            const statusCode = (lastDisconnect?.error instanceof Boom) ? lastDisconnect.error.output.statusCode : null;
            if (statusCode !== DisconnectReason.loggedOut) {
                botIniciado = false; // ✅ Permite reconexión
                setTimeout(iniciarBot, 3000);
            }
        } else if (connection === 'open') {
            console.clear();
            console.log('\n🚀 RADAR CUSCO CORREGIDO Y BLINDADO');
            console.log(`🎯 Zona Activa: ${CONFIG.ZONAS_CONFIG[estado.zonaPrioritaria].icono} ${CONFIG.ZONAS_CONFIG[estado.zonaPrioritaria].nombre}\n`);
        }
    });

    sock.ev.on('messages.upsert', async (m) => {
        if (m.type !== 'notify') return;
        
        for (const msg of m.messages) {
            if (!msg.message) continue;
            
            const jid = msg.key.remoteJid;
            const deMi = msg.key.fromMe;
            const msgId = msg.key.id;
            const ahora = Date.now();
            
            const { texto: cuerpoRaw, coords, mensajeReal } = extraerTextoYUbicacion(msg);
            const textoLimpio = normalizarTexto(cuerpoRaw);

            // ⚙️ COMANDOS PRIVADOS
            if (deMi) {
                const responderZona = async (key) => {
                    const z = CONFIG.ZONAS_CONFIG[key];
                    let text = `${z.icono} ZONA: ${z.nombre}`;
                    if (z.descripcion) text += `\n${z.descripcion}`;
                    text += `\n✅ Bot activado`;
                    await sock.sendMessage(jid, { text });
                };

                if (textoLimpio === '#wanchaq') {
                    estado.zonaPrioritaria = 'wanchaq'; estado.modoTodos = false; estado.zonaBaseParaTodos = null; estado.botActivo = true; estado.colaPedidos = []; estado.localAnclado = null; estado.subZonaActiva = null;
                    await responderZona('wanchaq');
                }
                else if (textoLimpio === '#sansebastian' || textoLimpio === '#cultura') {
                    estado.zonaPrioritaria = 'sansebastian'; estado.modoTodos = false; estado.zonaBaseParaTodos = null; estado.botActivo = true; estado.colaPedidos = []; estado.localAnclado = null; estado.subZonaActiva = null;
                    await responderZona('sansebastian');
                }
                else if (textoLimpio === '#centro') {
                    estado.zonaPrioritaria = 'centro'; estado.modoTodos = false; estado.zonaBaseParaTodos = null; estado.botActivo = true; estado.colaPedidos = []; estado.localAnclado = null; estado.subZonaActiva = null;
                    await responderZona('centro');
                }
                else if (textoLimpio === '#sanjeronimo' || textoLimpio === '#larapa' || textoLimpio === '#larafa') {
                    estado.zonaPrioritaria = 'sanjeronimo'; estado.modoTodos = false; estado.zonaBaseParaTodos = null; estado.botActivo = true; estado.colaPedidos = []; estado.localAnclado = null; estado.subZonaActiva = null;
                    await responderZona('sanjeronimo');
                }
                else if (textoLimpio === '#aeropuerto') {
                    estado.zonaPrioritaria = 'aeropuerto'; estado.modoTodos = false; estado.zonaBaseParaTodos = null; estado.botActivo = true; estado.colaPedidos = []; estado.localAnclado = null; estado.subZonaActiva = null;
                    await responderZona('aeropuerto');
                }
                else if (textoLimpio === '#todos') {
                    estado.zonaBaseParaTodos = estado.zonaPrioritaria;
                    estado.modoTodos = true; estado.botActivo = true; estado.colaPedidos = []; estado.localAnclado = null; estado.subZonaActiva = null;
                    await responderZona('todos');
                }
                else if (textoLimpio === '#ocupado') {
                    estado.botActivo = false;
                    await sock.sendMessage(jid, { text: '⏸️ Radar Pausado. Buen delivery!' });
                }
                else if (textoLimpio === '#libre') {
                    estado.botActivo = true; estado.colaPedidos = []; estado.localAnclado = null; estado.subZonaActiva = null;
                    await sock.sendMessage(jid, { text: `▶️ Radar Activo: ${CONFIG.ZONAS_CONFIG[estado.zonaPrioritaria].icono} ${CONFIG.ZONAS_CONFIG[estado.zonaPrioritaria].nombre}` });
                }
                else if (textoLimpio === '#limpiar' || textoLimpio === '#entregado') {
                    estado.colaPedidos = []; estado.localAnclado = null; estado.subZonaActiva = null;
                    await sock.sendMessage(jid, { text: '✅ COLA LIMPIADA' });
                }
                else if (textoLimpio === '#estado') {
                    let msg = '';
                    if (estado.modoTodos) {
                        msg = `${CONFIG.ZONAS_CONFIG.todos.icono} Modo: TODOS LOS LOCALES\n`;
                        if (estado.zonaBaseParaTodos) msg += `📍 Base: ${CONFIG.ZONAS_CONFIG[estado.zonaBaseParaTodos].nombre}\n`;
                    } else {
                        msg = `${CONFIG.ZONAS_CONFIG[estado.zonaPrioritaria].icono} Zona: ${CONFIG.ZONAS_CONFIG[estado.zonaPrioritaria].nombre}\n`;
                    }
                    msg += `🤖 Bot: ${estado.botActivo ? '✅ ACTIVO' : '⏸️ PAUSADO'}\n`;
                    msg += `📦 Cola: ${estado.colaPedidos.length}/${CONFIG.MAX_PEDIDOS_COLA}\n`;
                    if (estado.subZonaActiva) msg += `📍 Sub-zona: ${estado.subZonaActiva}\n`;
                    if (estado.localAnclado) msg += `⚓ Anclado a: ${estado.localAnclado.nombre.toUpperCase()}\n`;
                    await sock.sendMessage(jid, { text: msg });
                }
                else if (textoLimpio === '#cola') {
                    if (estado.colaPedidos.length === 0) {
                        await sock.sendMessage(jid, { text: '📦 Cola actualmente VACÍA.' });
                    } else {
                        let resumenCola = `📦 **DETALLE DE COLA (${estado.colaPedidos.length}/${CONFIG.MAX_PEDIDOS_COLA}):**\n\n`;
                        estado.colaPedidos.forEach((p, index) => {
                            const minTranscurridos = Math.round((ahora - p.timestamp) / 60000);
                            resumenCola += `${index + 1}. ${p.nombre.toUpperCase()} (${p.zona}) - hace ${minTranscurridos} min\n`;
                        });
                        await sock.sendMessage(jid, { text: resumenCola });
                    }
                }
                continue;
            }

            if (!estado.botActivo) continue;

            const esGrupo = jid.endsWith('@g.us');
            if (!esGrupo) continue;

            const nombreGrupoRaw = await obtenerNombreGrupo(sock, jid);
            const nombreGrupoLimpio = normalizarTexto(nombreGrupoRaw);

            if (CONFIG.GRUPOS_EXCLUIDOS.some(g => nombreGrupoLimpio.includes(g))) continue;

            if (textoLimpio === 'yo' && !deMi) {
                estado.gruposPedidoTomadoPorOtro.set(jid, ahora);
                continue;
            }
            if (estado.gruposPedidoTomadoPorOtro.has(jid)) continue;

            const tienePalabraInvalida = CONFIG.PALABRAS_ANTI_TRIGGER.some(p => textoLimpio.includes(p));
            if (tienePalabraInvalida) continue;

            const esUbicacion = !!mensajeReal.locationMessage || !!mensajeReal.liveLocationMessage || !!coords;
            const esImagen = !!mensajeReal.imageMessage;
            const esDocumento = !!mensajeReal.documentMessage;
            const tieneIntencionText = CONFIG.PALABRAS_INTENCION.some(p => textoLimpio.includes(p));

            if (estado.mensajesProcesados.has(msgId) || estado.idsEnProceso.has(msgId)) continue;
            if (estado.gruposEnEnfriamiento.has(jid)) {
                const ultimoYoEnGrupo = estado.gruposEnEnfriamiento.get(jid);
                if (ahora - ultimoYoEnGrupo < 15000) continue;
            }

            if (!esUbicacion && !esImagen && !esDocumento && !tieneIntencionText) continue;

            const zonaBase = estado.modoTodos ? (estado.zonaBaseParaTodos || estado.zonaPrioritaria) : estado.zonaPrioritaria;
            const coordsBase = getCoordenadasZona(zonaBase);

            let localEncontrado = null;
            let mejorScore = 0;
            let mejorLongitud = 0;

            for (const l of LOCALES) {
                if (!estado.modoTodos && l.zona !== estado.zonaPrioritaria) continue;

                if (coordsBase) {
                    const distBase = calcularDistancia(coordsBase.lat, coordsBase.lng, l.lat, l.lng);
                    if (distBase > CONFIG.DISTANCIA_MAX_TODOS) continue;
                }

                if (estado.subZonaActiva && l.subZona !== estado.subZonaActiva) {
                    const distSubZona = estado.colaPedidos.length > 0 ? 
                        calcularDistancia(estado.colaPedidos[0].lat, estado.colaPedidos[0].lng, l.lat, l.lng) : 0;
                    if (distSubZona > 1.5) continue;
                }

                let scoreActual = 0;
                let maxAliasLen = 0;

                const coincidenciaTexto = l.alias.some(alias => textoLimpio.includes(alias));
                const coincidenciaGrupo = l.alias.some(alias => nombreGrupoLimpio.includes(alias));

                if (coincidenciaTexto) {
                    scoreActual += 20;
                    const aliasEncontrado = l.alias.find(a => textoLimpio.includes(a));
                    if (aliasEncontrado) maxAliasLen = Math.max(maxAliasLen, aliasEncontrado.length);
                }

                if (coincidenciaGrupo && (esUbicacion || esImagen || esDocumento || tieneIntencionText)) {
                    scoreActual += 15;
                    const aliasEncontrado = l.alias.find(a => nombreGrupoLimpio.includes(a));
                    if (aliasEncontrado) maxAliasLen = Math.max(maxAliasLen, aliasEncontrado.length);
                }

                if (estado.subZonaActiva && l.subZona === estado.subZonaActiva) {
                    scoreActual += 10;
                }

                if (coords && coords.lat != null && coords.lng != null) {
                    const distPin = calcularDistancia(coords.lat, coords.lng, l.lat, l.lng);
                    if (distPin < 0.3) scoreActual += 25;
                }

                if (scoreActual > mejorScore || (scoreActual === mejorScore && maxAliasLen > mejorLongitud)) {
                    if (scoreActual >= 15) {
                        mejorScore = scoreActual;
                        mejorLongitud = maxAliasLen;
                        localEncontrado = l;
                    }
                }
            }

            if (!localEncontrado) continue;

            if (estado.localAnclado && (ahora - estado.tiempoAnclaje) / 60000 <= 15) {
                const esElMismo = localEncontrado.nombre === estado.localAnclado.nombre;
                const distAnclado = calcularDistancia(localEncontrado.lat, localEncontrado.lng, estado.localAnclado.lat, estado.localAnclado.lng);
                if (!esElMismo && distAnclado > 0.8) {
                    console.log(`🛑 Pedido ignorado por Anclaje en ${estado.localAnclado.nombre.toUpperCase()}`);
                    continue;
                }
            }

            if (estado.ultimoLocalDetectado === localEncontrado.nombre && (ahora - estado.tiempoUltimoLocal) < 20000) continue;

            estado.colaPedidos = estado.colaPedidos.filter(p => (ahora - p.timestamp) / 60000 <= CONFIG.MAX_MINUTOS_ACUMULADOS);

            if (estado.colaPedidos.length >= 2) {
                const local1 = estado.colaPedidos[0].nombre;
                const local2 = estado.colaPedidos[1].nombre;

                if (local1 !== local2) {
                    if (localEncontrado.nombre !== local1 && localEncontrado.nombre !== local2) {
                        continue;
                    }
                }

                if (estado.colaPedidos.length >= CONFIG.MAX_PEDIDOS_COLA) continue;
            }

            if (estado.colaPedidos.length === 0 && !estado.modoTodos) {
                estado.subZonaActiva = localEncontrado.subZona;
            }

            estado.idsEnProceso.add(msgId);
            estado.mensajesProcesados.add(msgId);
            estado.gruposEnEnfriamiento.set(jid, ahora);

            estado.localAnclado = localEncontrado;
            estado.tiempoAnclaje = ahora;

            sock.sendMessage(jid, { text: 'Yo' })
                .then(() => {
                    console.log(`⚡ [YO ENVIADO CON ÉXITO]: ${localEncontrado.nombre.toUpperCase()} | Grupo: ${nombreGrupoRaw}`);
                })
                .catch(e => console.log('❌ Error enviando Yo:', e.message));

            estado.ultimoLocalDetectado = localEncontrado.nombre;
            estado.tiempoUltimoLocal = ahora;
            const distanciaCasa = calcularDistancia(CONFIG.MI_UBICACION.lat, CONFIG.MI_UBICACION.lng, localEncontrado.lat, localEncontrado.lng);
            
            estado.colaPedidos.push({
                nombre: localEncontrado.nombre,
                zona: localEncontrado.zona,
                subZona: localEncontrado.subZona,
                lat: localEncontrado.lat,
                lng: localEncontrado.lng,
                timestamp: ahora
            });

            (async () => {
                try {
                    let msgNotif = `⚡ PEDIDO GANADO!\n\n👥 Grupo: ${nombreGrupoRaw}\n📍 ${localEncontrado.nombre.toUpperCase()}\n📏 Distancia: ${distanciaCasa} km\n🕐 Hora: ${formatearHora(ahora)}\n📦 Cola: ${estado.colaPedidos.length}/${CONFIG.MAX_PEDIDOS_COLA}\n`;
                    if (estado.subZonaActiva) msgNotif += `📍 Sub-zona: ${estado.subZonaActiva.toUpperCase()}\n`;
                    if (coords) msgNotif += `🌐 Coordenadas: ${coords.lat}, ${coords.lng}\n`;
                    if (estado.localAnclado) msgNotif += `⚓ Anclado a: ${estado.localAnclado.nombre.toUpperCase()}\n`;
                    await sock.sendMessage(CONFIG.MI_NUMERO, { text: msgNotif });
                } catch (e) {
                    console.log('Error en notificación privada:', e.message);
                }
            })();
        }
    });
}
