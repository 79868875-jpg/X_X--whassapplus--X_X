function calcularDistancia(lat1, lng1, lat2, lng2) {
    const R = 6371;
    const dLat = (lat2 - lat1) * Math.PI / 180;
    const dLng = (lng2 - lng1) * Math.PI / 180;
    const a = Math.sin(dLat/2) * Math.sin(dLat/2) +
              Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * 
              Math.sin(dLng/2) * Math.sin(dLng/2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
    return parseFloat((R * c).toFixed(2));
}

function getCoordenadasZona(zona) {
    const coords = {
        'centro': { lat: -13.5160, lng: -71.9800 },
        'wanchaq': { lat: -13.5180, lng: -71.9550 },
        'sansebastian': { lat: -13.5230, lng: -71.9450 },
        'sanjeronimo': { lat: -13.5070, lng: -71.9270 },
        'aeropuerto': { lat: -13.5357, lng: -71.9388 }
    };
    return coords[zona] || coords['aeropuerto'];
}

// ==========================================
// 🚀 NÚCLEO BAILEYS CORREGIDO
// ==========================================
async function iniciarBot() {
    console.log("🤖 Inicializando motor Baileys...");
    const { state, saveCreds } = await useMultiFileAuthState('sesion_baileys');
    
    const sock = makeWASocket({
        logger: pino({ level: 'silent' }), // Volvemos a silenciar los logs molestos
        auth: state,
        browser: ['Radar Cusco', 'Safari', '3.0']
    });

    sock.ev.on('creds.update', saveCreds);

    sock.ev.on('connection.update', (update) => {
        const { connection, lastDisconnect, qr } = update;
        
        // 📸 EVENTO CORRECTO PARA ENVIAR EL QR A LA PANTALLA
        if (qr) {
            console.clear();
            console.log('✨ [ESCANEAME] Escanea este código QR con tu WhatsApp para conectar el radar:\n');
            qrcode.generate(qr, { small: true });
        }

        if (connection === 'close') {
            const shouldReconnect = (lastDisconnect.error instanceof Boom) ? 
                lastDisconnect.error.output.statusCode !== DisconnectReason.loggedOut : true;
            if (shouldReconnect) iniciarBot();
        } else if (connection === 'open') {
            console.clear();
            console.log('\n🚀 RADAR CUSCO [BAILEYS] ACTIVO Y ESCANEADO');
            console.log(`🎯 Zona inicial: ${ZONAS_CONFIG[zonaPrioritaria].icono} ${ZONAS_CONFIG[zonaPrioritaria].nombre}\n`);
        }
    });

    sock.ev.on('messages.upsert', async (m) => {
        if (m.type !== 'notify') return;
        
        for (const msg of m.messages) {
            if (!msg.message) return;
            
            const jid = msg.key.remoteJid;
            const deMi = msg.key.fromMe;
            const msgId = msg.key.id;

            const textoMensaje = msg.message.conversation || 
                                 msg.message.extendedTextMessage?.text || 
                                 msg.message.imageMessage?.caption || "";
            const cuerpo = textoMensaje.toLowerCase().trim();

            if (deMi) {
                if (['#wanchaq', '#sansebastian', '#cultura', '#centro', '#sanjeronimo', '#larapa', '#aeropuerto'].some(cmd => cuerpo === cmd)) {
                    let mapeo = { '#cultura': 'sansebastian', '#larapa': 'sanjeronimo' };
                    let limpiaCmd = cuerpo.replace('#', '');
                    zonaPrioritaria = mapeo[cuerpo] || limpiaCmd;
                    modoTodos = false; botActivo = true; colaPedidos = []; rutaActiva = null; zonaBaseParaTodos = null;
                    await sock.sendMessage(jid, { text: `${ZONAS_CONFIG[zonaPrioritaria].icono} ZONA ACTIVA: ${ZONAS_CONFIG[zonaPrioritaria].nombre}\nCola vacía y listo.` });
                }
                else if (cuerpo === '#todos') {
                    modoTodos = true; botActivo = true; colaPedidos = []; rutaActiva = null; zonaBaseParaTodos = zonaPrioritaria;
                    await sock.sendMessage(jid, { text: `${ZONAS_CONFIG.todos.icono} MODO GLOBAL\n📍 Priorizando base: ${ZONAS_CONFIG[zonaPrioritaria].nombre}` });
                }
                else if (cuerpo === '#ocupado') { 
                    botActivo = false; 
                    await sock.sendMessage(jid, { text: '⏸️ Radar en PAUSA.' }); 
                }
                else if (cuerpo === '#libre') { 
                    botActivo = true; colaPedidos = []; rutaActiva = null; 
                    await sock.sendMessage(jid, { text: '▶️ Radar LIBRE buscando pedidos.' }); 
                }
                else if (cuerpo === '#limpiar' || cuerpo === '#entregado') { 
                    colaPedidos = []; rutaActiva = null; 
                    await sock.sendMessage(jid, { text: '✅ COLA LIMPIADA TOTALMENTE' }); 
                }
                return;
            }

            if (!botActivo) return;

            const esGrupo = jid.endsWith('@g.us');
            if (!esGrupo) return;

            if (mensajesProcesados.has(msgId) || idsEnProceso.has(msgId)) return;

            const nombreGrupoLimpio = (msg.pushName || "").toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
            const textoLimpio = cuerpo.normalize("NFD").replace(/[\u0300-\u036f]/g, "");
            const ahora = Date.now();

            const esUbicacion = msg.message.locationMessage || msg.message.liveLocationMessage;
            const tieneIntencionText = PALABRAS_INTENCION.some(p => textoLimpio.includes(p));

            if (!esUbicacion && !tieneIntencionText) return;

            let localEncontrado = null;

            for (const l of LOCALES) {
                if (esUbicacion) {
                    const locData = msg.message.locationMessage || msg.message.liveLocationMessage;
                    const distGps = calcularDistancia(locData.degreesLatitude, locData.degreesLongitude, l.lat, l.lng);
                    if (distGps < 0.3) { localEncontrado = l; break; }
                }

                const tieneAlias = l.alias ? l.alias.some(a => textoLimpio.includes(a) || nombreGrupoLimpio.includes(a)) : false;
                const tieneNombreDirecto = textoLimpio.includes(l.nombre) || nombreGrupoLimpio.includes(l.nombre);

                if (tieneAlias || tieneNombreDirecto) {
                    localEncontrado = l;
                    break;
                }
            }

            if (!localEncontrado) return;

            if (ultimoLocalDetectado === localEncontrado.nombre && (ahora - tiempoUltimoLocal) < 25000) return;

            const coordsBase = modoTodos ? getCoordenadasZona(zonaBaseParaTodos) : getCoordenadasZona(zonaPrioritaria);
            const distanciaDesdeBase = calcularDistancia(coordsBase.lat, coordsBase.lng, localEncontrado.lat, localEncontrado.lng);
            const distanciaDesdeAeropuerto = calcularDistancia(MI_UBICACION.lat, MI_UBICACION.lng, localEncontrado.lat, localEncontrado.lng);

            if (!modoTodos) {
                if (localEncontrado.zona !== zonaPrioritaria) return;
            } else {
                if (distanciaDesdeBase > 3.0) return;
            }

            if (colaPedidos.length >= MAX_PEDIDOS_COLA) {
                if ((ahora - colaPedidos[0].timestamp) / 60000 > MAX_MINUTOS_ACUMULADOS) {
                    colaPedidos.shift();
                } else {
                    return;
                }
            }

            if (colaPedidos.length > 0) {
                const primerPedido = colaPedidos[0];
                if (colaPedidos.length === 1) {
                    if (localEncontrado.nombre !== primerPedido.nombre) return;
                }
                else if (colaPedidos.length === 2) {
                    if (localEncontrado.zona !== zonaPrioritaria) return;
                }
            }

            idsEnProceso.add(msgId);
            mensajesProcesados.add(msgId);
            setTimeout(() => idsEnProceso.delete(msgId), 5000);
            setTimeout(() => mensajesProcesados.delete(msgId), 60000);

            await sock.sendMessage(jid, { text: 'Yo' }, { quoted: msg });

            ultimoLocalDetectado = localEncontrado.nombre;
            tiempoUltimoLocal = ahora;

            colaPedidos.push({ 
                nombre: localEncontrado.nombre, 
                zona: localEncontrado.zona, 
                lat: localEncontrado.lat, 
                lng: localEncontrado.lng, 
                timestamp: ahora 
            });

            if (colaPedidos.length === 1) {
                if (ZONAS_SUBIDA.some(z => textoLimpio.includes(z))) rutaActiva = 'SUBIDA';
                else if (ZONAS_BAJADA.some(z => textoLimpio.includes(z))) rutaActiva = 'BAJADA';
                else rutaActiva = (zonaPrioritaria === 'centro' || zonaPrioritaria === 'wanchaq') ? 'SUBIDA' : 'BAJADA';
            }

            console.log(`🚀 [Baileys] YO enviado a: ${localEncontrado.nombre.toUpperCase()}`);

            try {
                let msgNotificacion = `⚡ ¡PEDIDO GANADO!\n\n🏪 ${localEncontrado.nombre.toUpperCase()}\n📍 Zona: ${ZONAS_CONFIG[localEncontrado.zona].icono} ${localEncontrado.zona.toUpperCase()}\n✈️ Dist. Aeropuerto: ${distanciaDesdeAeropuerto} km\n📦 Cola: ${colaPedidos.length}/${MAX_PEDIDOS_COLA}`;
                if (rutaActiva) msgNotificacion += `\n🛵 Dirección: ${rutaActiva}`;
                
                await sock.sendMessage(MI_NUMERO, { text: msgNotificacion });
            } catch (e) {
                console.log("Error en reporte privado:", e.message);
            }
        }
    });
}

iniciarBot();
